# What an auction-only order cannot be: entered during continuous trading, a market order, an
# immediacy demand, or a stop, because it exists to rest until its call.
INSTRUMENT tick=5 lot=1 min=1 max=1000000 band=500 open=100000 alloc=PRICE_TIME
SESSION CONTINUOUS
STATE CONTINUOUS
NEW BUY LIMIT GTC 100000 10 AUCTION_ONLY #1
REJECTED #1 STATE_NOT_PERMITTED
SESSION CLOSING_AUCTION
STATE CLOSING_AUCTION
NEW BUY MARKET IOC 0 10 AUCTION_ONLY #2
REJECTED #2 INVALID_FIELDS
NEW BUY LIMIT IOC 100000 10 AUCTION_ONLY #3
REJECTED #3 INVALID_FIELDS
NEW BUY LIMIT GTC 99000 10 trigger=99500 AUCTION_ONLY #4
REJECTED #4 INVALID_FIELDS
NEW BUY LIMIT GTC 100000 10 AUCTION_ONLY #5
ACCEPTED #5 order=1
RESTED order=1 BUY 100000 10
SESSION CLOSED
STATE CLOSED
REMOVED order=1 10 EXPIRED
